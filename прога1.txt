			for (c = 0; c < 3; c++)

				for (sq = 0; sq < 4; sq++)
				{
					for (i = 0; i < R; i++)
						for (j = 0; j < R; j++)
						{
							value = PicGet(PSrc, x + j - (sq == 1 || sq == 2 ? R : -1),
							y + i - (sq == 2 || sq == 4 ? R : -1), c);
							sum[sq] += value;
							dis[sq] += value * value;
						}
						dis[sq] = dis[sq] / (n - 1) - sum[sq] * sum[sq] / (n * n - n);
				}
				min = dis[0], min_i = 0;
				for (i = 1; i < 4; i++)
				if (min > dis[i])
				min = dis[i], min_i = i;

				PDest->Pixels[y * PDest->W + x][c] = Clamp((int)sum[min_i] / (R * R), 0, 255);

				memset(sum, 0, 4 * sizeof(double));
				memset(dis, 0, 4 * sizeof(double));
			}
